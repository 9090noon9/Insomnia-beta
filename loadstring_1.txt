local NamecallDispatcher = { Hooks = {}, Original = nil }
function NamecallDispatcher:Register(cb) self.Hooks[#self.Hooks + 1] = cb; return #self.Hooks end
function NamecallDispatcher:CallOriginal(obj, ...) if self.Original then return self.Original(obj, ...) end end

pcall(function()
    NamecallDispatcher.Original = hookmetamethod(game, "__namecall", newcclosure(function(object, ...)
        local count = #NamecallDispatcher.Hooks
        if count == 0 then return NamecallDispatcher.Original(object, ...) end
        local method = getnamecallmethod()
        for i = 1, count do
            local result = NamecallDispatcher.Hooks[i](object, method, ...)
            if result ~= nil and result ~= false then
                if result == true then return nil end
                return result
            end
        end
        return NamecallDispatcher.Original(object, ...)
    end))
end)

local function CloneCosmetic(name, cosmetic_type, options)
    if not modules.CosmeticLibrary or not modules.CosmeticLibrary.Cosmetics then return nil end
    local base = modules.CosmeticLibrary.Cosmetics[name]
    if not base then return nil end
    local data = table.clone(base)
    data.Name = name
    data.Type = data.Type or cosmetic_type
    data.Seed = math.random(1, 1000000)
    if options then
        if options.inverted then data.Inverted = true end
        if options.favorites_only then data.OnlyUseFavorites = true end
    end
    return data
end

local function HandleSkinchangerEquip(args)
    local weapon_name = args[1]
    local cosmetic_type = args[2]
    local cosmetic_name = args[3]
    local options = args[4] or {}
    states.skinchanger_state.equipped[weapon_name] = states.skinchanger_state.equipped[weapon_name] or {}
    if not cosmetic_name or cosmetic_name == "" or cosmetic_name == "None" or cosmetic_name == "NONE_COSMETIC" then
        states.skinchanger_state.equipped[weapon_name][cosmetic_type] = { Name = "NONE_COSMETIC" }
    elseif cosmetic_name == "Random" or cosmetic_name == "RANDOM_COSMETIC" then
        states.skinchanger_state.equipped[weapon_name][cosmetic_type] = {
            Name = cosmetic_name, Type = cosmetic_type,
            Inverted = options.IsInverted, OnlyUseFavorites = options.OnlyUseFavorites,
            Seed = math.random(1, 1000000),
        }
    else
        local cloned = CloneCosmetic(cosmetic_name, cosmetic_type, {
            inverted = options.IsInverted, favorites_only = options.OnlyUseFavorites,
        })
        if cloned then states.skinchanger_state.equipped[weapon_name][cosmetic_type] = cloned end
    end
    task.spawn(function()
        task.wait(0.1)
        pcall(function()
            if modules.PlayerDataController and modules.PlayerDataController.CurrentData then
                modules.PlayerDataController.CurrentData:Replicate("WeaponInventory")
            end
        end)
    end)
    task.spawn(function()
        task.wait(0.2)
        pcall(function()
            local lf = FighterController and FighterController.LocalFighter
            if not lf or not lf.EquippedItem then return end
            local item = lf.EquippedItem
            local vm = item.ViewModel
            if vm and vm.Destroy then vm:Destroy() end
            item.ViewModel = nil
            if item._CreateViewModel then
                item.ViewModel = item:_CreateViewModel(item.ViewModelRef)
            end
        end)
    end)
    pcall(SaveEquipped)
end

        if method ~= "FireServer" then return false end
        local args = {...}
        if self.Name == "EquipCosmetic" then

local function ResolveCosmetic(weaponName, cosmeticType)
    local equipped = states.skinchanger_state.equipped[weaponName]
        and states.skinchanger_state.equipped[weaponName][cosmeticType]
    if not equipped then return nil end
    if equipped.Name == "None" or equipped.Name == "NONE_COSMETIC" then return nil end
    return equipped
end

                if weaponName and weaponPlayer == LocalPlayer and states.skinchanger_state.equipped[weaponName] and ResolveCosmetic(weaponName, "Wrap") then
                    return ResolveCosmetic(weaponName, "Wrap")
                end
                return origGetWrap(self)
            end
        end

        local origNew = ClientViewModel2.new
        ClientViewModel2.new = function(replicatedData, clientItem)
            local weaponPlayer = clientItem.ClientFighter and clientItem.ClientFighter.Player
            local weaponName = states.skinchanger_state.constructing_weapon or clientItem.Name

            if weaponPlayer == LocalPlayer and states.skinchanger_state.equipped[weaponName] then
                pcall(function()
                    local ReplicatedClass = require(ReplicatedStorage.Modules.ReplicatedClass)
                    local dataKey = ReplicatedClass:ToEnum("Data")
                    replicatedData[dataKey] = replicatedData[dataKey] or {}

                    if ResolveCosmetic(weaponName, "Skin") then
                        replicatedData[dataKey][ReplicatedClass:ToEnum("Skin")] = ResolveCosmetic(weaponName, "Skin")
                    end
                    if ResolveCosmetic(weaponName, "Wrap") then
                        replicatedData[dataKey][ReplicatedClass:ToEnum("Wrap")] = ResolveCosmetic(weaponName, "Wrap")
                    end
                    if ResolveCosmetic(weaponName, "Charm") then
                        replicatedData[dataKey][ReplicatedClass:ToEnum("Charm")] = ResolveCosmetic(weaponName, "Charm")
                    end
                end)
            end

            local result = origNew(replicatedData, clientItem)

            pcall(function()
                local objectID = nil
                pcall(function()
                    local ReplicatedClass = require(ReplicatedStorage.Modules.ReplicatedClass)
                    local dataKey = ReplicatedClass:ToEnum("Data")
                    local objKey = ReplicatedClass:ToEnum("ObjectID")
                    if replicatedData[dataKey] then
                        objectID = replicatedData[dataKey][objKey]
                    end
                end)
                if objectID == nil then
                    pcall(function() objectID = result:Get("ObjectID") or result.ObjectID end)
                end
                if objectID and weaponPlayer == LocalPlayer and states.skinchanger_state.equipped[weaponName] then
                    states.skinchanger_state.placed_object_map[objectID] = weaponName
                end
            end)

            if weaponPlayer == LocalPlayer and states.skinchanger_state.equipped[weaponName] and ResolveCosmetic(weaponName, "Wrap") and result._UpdateWrap then
                task.spawn(function()
                    result:_UpdateWrap()
                    task.wait(0.1)
                    if not result._destroyed then result:_UpdateWrap() end
                end)
            end

            return result
        end
    end)
end)

local function GetAllCosmetics()
    local results = {}
    if not modules.CosmeticLibrary or not modules.CosmeticLibrary.Cosmetics then return results end
    for name, _ in pairs(modules.CosmeticLibrary.Cosmetics) do table.insert(results, name) end
    return results
end

local function UnlockAllCosmetics()
    local list = GetAllCosmetics()
    for _, cosmetic in ipairs(list) do
        if not cosmetic:find("MISSING_") then states.skinchanger_state.fake_owned[cosmetic] = true end
    end
    pcall(function()
        if modules.PlayerDataController and modules.PlayerDataController.CurrentData then
            modules.PlayerDataController.CurrentData:Replicate("CosmeticInventory")
        end
    end)
end

local function UnlockAllWeapons()
    local current_data = modules.PlayerDataController and modules.PlayerDataController.CurrentData
    if not current_data then return end
    local inventory = current_data:Get("WeaponInventory")
    local owned = {}
    if type(inventory) == "table" then
        for _, weapon_data in pairs(inventory) do
            if type(weapon_data) == "table" and weapon_data.Name then owned[weapon_data.Name] = true end
        end
    end
    if not modules.ShopLibrary or not modules.ShopLibrary.GetReleasedOwnableWeapons then return end
    local ownables = modules.ShopLibrary:GetReleasedOwnableWeapons()
    for _, weapon_name in pairs(ownables) do
        if not owned[weapon_name] and not states.skinchanger_state.fake_weapon_names[weapon_name] then
            table.insert(states.skinchanger_state.fake_weapon_owned, { Name = weapon_name, Level = 1, XP = 0, IsFavorited = false, Skin = nil })
            states.skinchanger_state.fake_weapon_names[weapon_name] = true
        end
    end
    pcall(function()
        if modules.PlayerDataController and modules.PlayerDataController.CurrentData then
            modules.PlayerDataController.CurrentData:Replicate("WeaponInventory")
        end
    end)
end

    UnlockAllCosmetics()
    UnlockAllWeapons()
end

local function MakeGradientSequence(cA, cB, cC)
    return ColorSequence.new({
        ColorSequenceKeypoint.new(0, cA),
        ColorSequenceKeypoint.new(0.5, cB),
        ColorSequenceKeypoint.new(1, cC),
    })
end

local function ApplyGradient(gradient, cfg, prefix, spinning)
    local cA = cfg[prefix .. "ColorStart"] or WHITE
    local cB = cfg[prefix .. "ColorMiddle"] or cA
    local cC = cfg[prefix .. "ColorEnd"] or cB
    gradient.Color = MakeGradientSequence(cA, cB, cC)
    if spinning then
        gradient.Rotation = (RotationState.Time * (cfg[prefix .. "SpinSpeed"] or 1) * 60) % 360
    else
        gradient.Rotation = 45
    end
end

local function BuildFOVCircle(name, parent, thickness, filled, alpha)
    local frame = Instance.new("Frame")
    frame.Name = name
    frame.AnchorPoint = Vector2.new(0.5, 0.5)
    frame.BackgroundTransparency = 1
    frame.BorderSizePixel = 0
    frame.Visible = false
    frame.ZIndex = 2
    frame.Parent = parent
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(1, 0)
    corner.Parent = frame
    if filled then
        frame.BackgroundColor3 = WHITE
        frame.BackgroundTransparency = 1 - alpha
        local grad = Instance.new("UIGradient")
        grad.Parent = frame
        return frame, nil, grad
    else
        local stroke = Instance.new("UIStroke")
        stroke.Thickness = thickness
        stroke.Color = WHITE
        stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        stroke.Parent = frame
        local grad = Instance.new("UIGradient")
        grad.Parent = stroke
        return frame, stroke, grad
    end
end

local function UpdateFOV()
    local center = Vector2.new(Camera.ViewportSize.X * 0.5, Camera.ViewportSize.Y * 0.5)
    if Config.AimEnabled and Config.FOVEnabled then
        FOVFrame.Visible = true
        FOVFrame.Position = UDim2.fromOffset(center.X, center.Y)
        FOVFrame.Size = UDim2.fromOffset(Config.FOVRadius * 2, Config.FOVRadius * 2)
        ApplyGradient(FOVGradient, Config, "FOV", Config.FOVSpin)
        if Config.FOVFill then
            FOVFillFrame.Visible = true
            FOVFillFrame.Position = UDim2.fromOffset(center.X, center.Y)
            FOVFillFrame.Size = UDim2.fromOffset(Config.FOVRadius * 2, Config.FOVRadius * 2)
            FOVFillFrame.BackgroundTransparency = 1 - Config.FOVFillAlpha
            ApplyGradient(FOVFillGradient, Config, "FOVFill", Config.FOVFillSpin)
        else
            FOVFillFrame.Visible = false
        end
    else
        FOVFrame.Visible = false
        FOVFillFrame.Visible = false
    end
    if Config.AimbotEnabled and Config.AimbotShowFOV then
        AimbotFOVFrame.Visible = true
        AimbotFOVFrame.Position = UDim2.fromOffset(center.X, center.Y)
        AimbotFOVFrame.Size = UDim2.fromOffset(Config.AimbotFOV * 2, Config.AimbotFOV * 2)
        ApplyGradient(AimbotFOVGradient, Config, "AimbotFOV", Config.AimbotFOVSpin)
        if Config.AimbotFOVFill then
            AimbotFOVFillFrame.Visible = true
            AimbotFOVFillFrame.Position = UDim2.fromOffset(center.X, center.Y)
            AimbotFOVFillFrame.Size = UDim2.fromOffset(Config.AimbotFOV * 2, Config.AimbotFOV * 2)
            AimbotFOVFillFrame.BackgroundTransparency = 1 - Config.AimbotFOVFillAlpha
            ApplyGradient(AimbotFOVFillGradient, Config, "AimbotFOVFill", Config.AimbotFOVFillSpin)
        else
            AimbotFOVFillFrame.Visible = false
        end
    else
        AimbotFOVFrame.Visible = false
        AimbotFOVFillFrame.Visible = false
    end
end

local function getOrCreate(class, name)
    local obj = Lighting:FindFirstChild(name)
    if not obj then
        obj = Instance.new(class)
        obj.Name = name
        obj.Parent = Lighting
    end
    return obj
end

local function applyWorld()
    Lighting.Ambient = Config.WorldAmbient and Config.WorldAmbientColor or WorldDefaults.Ambient
    Lighting.OutdoorAmbient = Config.WorldOutdoorAmbient and Config.WorldOutdoorAmbientColor or WorldDefaults.OutdoorAmbient
    Lighting.Brightness = Config.WorldBrightness and Config.WorldBrightnessValue or WorldDefaults.Brightness
    Lighting.ClockTime = Config.WorldClockTime and Config.WorldClockTimeValue or WorldDefaults.ClockTime
    if Config.WorldFog then
        Lighting.FogColor = Config.WorldFogColor
        Lighting.FogStart = Config.WorldFogStart
        Lighting.FogEnd = Config.WorldFogEnd
    else
        Lighting.FogColor = WorldDefaults.FogColor
        Lighting.FogStart = WorldDefaults.FogStart
        Lighting.FogEnd = WorldDefaults.FogEnd
    end
    Lighting.ExposureCompensation = Config.WorldExposure and Config.WorldExposureValue or WorldDefaults.ExposureCompensation
    for _, sky in ipairs(Lighting:GetChildren()) do
        if sky:IsA("Sky") and sky.Name ~= "insomniaSky" then sky:Destroy() end
    end
    local sky = Lighting:FindFirstChild("insomniaSky")
    if Config.WorldSkybox then
        if not sky then
            sky = Instance.new("Sky")
            sky.Name = "insomniaSky"
            sky.Parent = Lighting
        end
        local preset = SkyPresets[Config.WorldSkyboxPreset]
        if preset then
            pcall(function()
                sky.SkyboxBk = preset.Bk
                sky.SkyboxDn = preset.Dn
                sky.SkyboxFt = preset.Ft
                sky.SkyboxLf = preset.Lf

                sky.SkyboxUp = preset.Up
                sky.SunTextureId = ""
                sky.MoonTextureId = ""
                sky.StarCount = 0
            end)
        end
    else
        if sky then sky:Destroy() end
    end
    if Config.WorldBloom then
        local bloom = getOrCreate("BloomEffect", "insomniaBloom")
        bloom.Intensity = Config.WorldBloomIntensity
        bloom.Size = 24
        bloom.Threshold = 0.95
    else
        local b = Lighting:FindFirstChild("insomniaBloom")
        if b then b:Destroy() end
    end
    if Config.WorldColorCorrection then
        local cc = getOrCreate("ColorCorrectionEffect", "insomniaCC")
        cc.TintColor = Config.WorldColorCorrectionColor
    else
        local cc = Lighting:FindFirstChild("insomniaCC")
        if cc then cc:Destroy() end
    end
    if Config.WorldSaturation then
        local sat = getOrCreate("ColorCorrectionEffect", "insomniaSat")
        sat.Saturation = Config.WorldSaturationValue
    else
        local sat = Lighting:FindFirstChild("insomniaSat")
        if sat then sat:Destroy() end
    end
    if Config.WorldContrast then
        local con = getOrCreate("ColorCorrectionEffect", "insomniaCon")
        con.Contrast = Config.WorldContrastValue
    else
        local con = Lighting:FindFirstChild("insomniaCon")
        if con then con:Destroy() end
    end
    if Config.WorldAtmosphere then
        local atm = getOrCreate("Atmosphere", "insomniaAtm")
        atm.Density = Config.WorldAtmosphereDensity
        atm.Haze = Config.WorldAtmosphereHaze
        atm.Glare = Config.WorldAtmosphereGlare

        sr.Intensity = Config.WorldSunRaysIntensity

local WeatherFolder = nil
local WeatherConnection = nil
local WeatherSound = nil
local WeatherParticles = {}
local WeatherParticleData = {}

local function clearWeather()
    if WeatherConnection then WeatherConnection:Disconnect(); WeatherConnection = nil end
    if WeatherSound then WeatherSound:Stop(); WeatherSound:Destroy(); WeatherSound = nil end
    for _, obj in ipairs(workspace:GetChildren()) do
        if obj.Name == "insomniaWeather" then obj:Destroy() end
    end
    WeatherFolder = nil
    WeatherParticles = {}
    WeatherParticleData = {}
end

local function applyWeather()
    clearWeather()
    if not Config.WeatherEnabled then return end
    if Config.WeatherType == "None" then return end
    local preset = WeatherConfigs[Config.WeatherType]
    if not preset then return end
    local intMul = (Config.WeatherIntensity or 50) / 50
    local count = math.floor(preset.Count * intMul)
    local area = preset.Area
    local height = preset.Height
    local speed = preset.Speed
    local mode = preset.Mode
    local folder = Instance.new("Folder")
    folder.Name = "insomniaWeather"
    folder.Parent = workspace
    WeatherFolder = folder
    local particles = {}
    local data = {}
    for i = 1, count do
        local part = Instance.new("Part")
        part.Name = "Particle_" .. i
        part.Shape = preset.Shape or Enum.PartType.Block
        part.Size = preset.Size
        part.Anchored = true
        part.CanCollide = false
        part.CanQuery = false
        part.CanTouch = false
        part.CastShadow = false
        part.Material = preset.Material or Enum.Material.SmoothPlastic
        part.Color = preset.Color
        part.Transparency = preset.Transparency or 0
        part.Parent = folder
        if preset.RainbowColor then part.Color = Color3.fromHSV(math.random(), 1, 1) end
        local offX = math.random(-area/2, area/2)
        local offZ = math.random(-area/2, area/2)
        local offY = math.random(0, height)
        particles[i] = part
        data[i] = { x = offX, y = offY, z = offZ, rot = math.random(0, 360) }
    end
    WeatherParticles = particles
    WeatherParticleData = data
    local presetRef = preset
    local modeRef = mode
    local speedRef = speed
    local function getPlayerRootPos()
        local char = LocalPlayer.Character
        if char then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then return root.Position end
        end
        return Vector3.new(0, 0, 0)
    end
    WeatherConnection = RunService.RenderStepped:Connect(function(dt)
        if not WeatherFolder or not WeatherFolder.Parent then
            if WeatherConnection then WeatherConnection:Disconnect() end
            return
        end
        local rootPos = getPlayerRootPos()
        local dtSpeed = speedRef * dt
        for i, drop in ipairs(particles) do
            if not drop.Parent then return end
            local d = data[i]
            if not d then return end
            if modeRef == "fall" then
                d.y = d.y - dtSpeed
                if presetRef.Drift and presetRef.Drift > 0 then
                    d.x = d.x + math.sin(tick() + i) * presetRef.Drift * dt
                end
            elseif modeRef == "rise" then
                d.y = d.y - dtSpeed
            elseif modeRef == "float" then
                d.x = d.x + math.sin(tick() * 0.8 + i) * 5 * dt
                d.z = d.z + math.cos(tick() * 0.6 + i) * 5 * dt
                d.y = d.y + math.sin(tick() * 0.4 + i) * 2 * dt
            end
            local reset = false
            if modeRef == "fall" and d.y < -10 then reset = true end
            if modeRef == "rise" and d.y > height then reset = true end
            if reset then
                d.x = math.random(-area/2, area/2)
                d.z = math.random(-area/2, area/2)
                if modeRef == "fall" then d.y = height
                elseif modeRef == "rise" then d.y = 0
                else d.y = math.random(0, height) end
            end
            drop.Position = Vector3.new(rootPos.X + d.x, rootPos.Y + d.y, rootPos.Z + d.z)
            if presetRef.RotSpeed and presetRef.RotSpeed > 0 then
                d.rot = d.rot + presetRef.RotSpeed * dt
                drop.CFrame = CFrame.new(drop.Position) * CFrame.Angles(math.rad(d.rot), math.rad(d.rot * 0.7), math.rad(d.rot * 0.5))
            end
        end
    end)
    if preset.Sound then
        local sound = Instance.new("Sound")
        sound.Name = "insomniaWeatherSound"
        sound.SoundId = preset.Sound
        sound.Looped = true
        sound.Volume = preset.SoundVolume or 0.5
        sound.Parent = SoundService
        sound:Play()
        WeatherSound = sound
    end
end

local function GetAimPart(Entity)
    if not Entity then return nil end
    if Config.ClosestPart then
        local Mouse = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        local Best, BestDist = nil, math.huge
        for _, Name in ipairs(HitboxNames) do
            local Part = Entity:FindFirstChild(Name, true)
            if Part and Part:IsA("BasePart") then
                local sp, onScreen = Camera:WorldToViewportPoint(Part.Position)
                if onScreen then
                    local d = (Vector2.new(sp.X, sp.Y) - Mouse).Magnitude
                    if d < BestDist then Best, BestDist = Part, d end
                end
            end
        end
        return Best
    end
    if Config.HitPart and Config.HitPart ~= "" then
        local Part = Entity:FindFirstChild(Config.HitPart, true)
        if Part and Part:IsA("BasePart") then return Part end
    end
    for _, Name in ipairs(HitboxNames) do
        local Part = Entity:FindFirstChild(Name, true)
        if Part and Part:IsA("BasePart") then return Part end
    end
    return nil
end

local function GetAimPosition(Part)
    if not Part then return nil end
    if Config.ClosestPos then
        local Origin = Camera.CFrame.Position
        local Dir = Part.Position - Origin
        local Params = RaycastParams.new()
        Params.FilterType = Enum.RaycastFilterType.Include
        Params.FilterDescendantsInstances = {Part}
        local Result = workspace:Raycast(Origin, Dir, Params)
        if Result then return Result.Position end
    end
    return Part.Position
end

local WallParams = RaycastParams.new()
WallParams.FilterType = Enum.RaycastFilterType.Exclude
WallParams.IgnoreWater = true

local function HasLineOfSight(Part)
    local Origin = Camera.CFrame.Position
    local Direction = Part.Position - Origin
    local Filter = {}
    if LocalPlayer.Character then
        for _, d in ipairs(LocalPlayer.Character:GetDescendants()) do
            if d:IsA("BasePart") then table.insert(Filter, d) end
        end
    end
    WallParams.FilterDescendantsInstances = Filter
    local Result = workspace:Raycast(Origin, Direction, WallParams)
    if Result then
        if Result.Instance:IsDescendantOf(Part.Parent) then return true end
        return false
    end
    return true
end

local function IsProtected(Entity)
    local ok, val = pcall(function() return Entity:GetAttribute("IsInvincible") end)
    if ok and val then return true end
    pcall(function()
        local plr = Players:GetPlayerFromCharacter(Entity)
        if plr and FighterController and FighterController.GetFighter then
            local fighter = FighterController:GetFighter(plr)
            if fighter then
                local ok2, v = pcall(function() return fighter.Entity:Get("IsInvincible") end)
                if ok2 and v then val = true end
            end
        end
    end)
    return val == true
end

local function IsFlashing()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if pg and pg:FindFirstChild("FlashbangGui") then return true end
    for _, v in ipairs(workspace:GetChildren()) do
        if v.Name == "FlashbangEffect" then return true end
    end
    return false
end

local function TeamCheck(Entity)
    local Player = Players:GetPlayerFromCharacter(Entity)
    if not Player then return true end
    local MyTeam = LocalPlayer:GetAttribute("TeamID")
    local TheirTeam = Player:GetAttribute("TeamID")
    if MyTeam ~= nil and TheirTeam ~= nil and MyTeam == TheirTeam then return false end
    return true
end

local function PassesCommonChecks(Entity, Part, fovRadius, opts)
    if not Part then return false end
    if not TeamCheck(Entity) then return false end
    local Player = Players:GetPlayerFromCharacter(Entity)
    if Player then
        local Humanoid = Entity:FindFirstChildOfClass("Humanoid")
        if not Humanoid or Humanoid.Health <= 0 then return false end
    end
    if opts.IgnoreProtected and IsProtected(Entity) then return false end
    if opts.AntiKatana and is_deflecting(Player) then return false end
    if opts.RiotShield and has_riot(Player) then return false end
    if opts.VisibleOnly and not HasLineOfSight(Part) then return false end
    if opts.DisableOnFlash and IsFlashing() then return false end
    local ScreenPoint, OnScreen = Camera:WorldToViewportPoint(Part.Position)
    if not OnScreen then return false end
    local ScreenDistance = (Vector2.new(ScreenPoint.X, ScreenPoint.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).Magnitude
    if fovRadius and ScreenDistance > fovRadius then return false end
    local WorldDistance = (Part.Position - Camera.CFrame.Position).Magnitude
    if opts.LimitDistance and WorldDistance > opts.MaxDistance then return false end
    return true, ScreenDistance, WorldDistance
end

local silentLastTarget = nil
local silentLastSeen = 0

local function SilentAimTarget()
    if not Config.AimEnabled then return nil end
    local opts = {
        VisibleOnly = Config.SilentVisibleOnly,
        IgnoreProtected = Config.SilentIgnoreProtected,
        AntiKatana = Config.SilentAntiKatana,
        RiotShield = Config.SilentRiotShield,
        DisableOnFlash = Config.SilentDisableOnFlash,
        LimitDistance = Config.SilentLimitDistance,
        MaxDistance = Config.AimbotMaxDistance,
    }
    if silentLastTarget and silentLastTarget.Parent and (os.clock() - silentLastSeen) < Config.AimbotForgetTime then
        local Part = GetAimPart(silentLastTarget)
        local passes = PassesCommonChecks(silentLastTarget, Part, Config.FOVRadius, opts)
        if passes then
            silentLastSeen = os.clock()
            return Part
        end
    end
    local Best, BestScreen, BestWorld = nil, math.huge, math.huge
    for _, Entity in CollectionService:GetTagged("Entity") do
        if Entity == LocalPlayer.Character then continue end
        local Part = GetAimPart(Entity)
        local passes, sd, wd = PassesCommonChecks(Entity, Part, Config.FOVRadius, opts)
        if not passes then continue end
        if sd + 15 < BestScreen or (math.abs(sd - BestScreen) <= 15 and wd < BestWorld) then
            Best, BestScreen, BestWorld = Part, sd, wd
            silentLastTarget = Entity
        end
    end
    if Best then silentLastSeen = os.clock() end
    return Best
end

local function AimbotTarget()
    if not Config.AimbotEnabled then return nil end
    local opts = {
        VisibleOnly = Config.AimbotVisibleOnly,
        IgnoreProtected = Config.AimbotIgnoreProtected,
        AntiKatana = Config.AimbotAntiKatana,
        RiotShield = Config.AimbotRiotShield,
        DisableOnFlash = Config.AimbotDisableOnFlash,
        LimitDistance = Config.AimbotLimitDistance,
        MaxDistance = Config.AimbotMaxDistance,
    }
    local Mouse = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local Best, BestScreen = nil, math.huge
    for _, Entity in CollectionService:GetTagged("Entity") do
        if Entity == LocalPlayer.Character then continue end
        local Part
        if Config.AimbotClosestPart then
            local bd = math.huge
            for _, Name in ipairs(HitboxNames) do
                local p = Entity:FindFirstChild(Name, true)
                if p and p:IsA("BasePart") then
                    local sp = Camera:WorldToViewportPoint(p.Position)
                    local d = (Vector2.new(sp.X, sp.Y) - Mouse).Magnitude
                    if d < bd then bd = d; Part = p end
                end
            end
        else
            Part = Entity:FindFirstChild(Config.AimbotHitPart, true) or Entity:FindFirstChild("Head", true)
        end
        local passes, sd = PassesCommonChecks(Entity, Part, Config.AimbotFOV, opts)
        if not passes then continue end
        if sd < BestScreen then Best = Part; BestScreen = sd end
    end
    return Best
end

local function EquippedInfo()
    local Fighter = FighterController and FighterController.LocalFighter
    local Item = Fighter and Fighter.EquippedItem
    return Item and Item.Info
end

local function Grappler()
    local Info = EquippedInfo()
    return Info ~= nil and Info.Name == "Grappler"
end

local function BulletDrop(Distance)
    local Info = EquippedInfo()
    local Speed = Info and Info.ProjectileSpeed
    local Gravity = Info and Info.ProjectileGravity
    if Speed and Speed > 0 and Gravity and Gravity > 0 then
        local Time = Distance / Speed
        return Gravity * workspace.Gravity * 0.5 * Time * Time
    end
    return 0
end

local function GetPartPosition(Char, Name)
    local P = Char:FindFirstChild(Name)
    if P and P:IsA("BasePart") then return P.Position end
    if Name == "Head" then
        for _, hn in ipairs(HitboxNames) do
            local h = Char:FindFirstChild(hn, true)

        end

local function WorldToScreen(Pos)
    local sp, onScreen = Camera:WorldToViewportPoint(Pos)
    return Vector2.new(sp.X, sp.Y), onScreen, sp.Z
end

local function PositionLine(line, from, to, thickness)
    local diff = to - from
    local length = diff.Magnitude
    if length < 0.5 then line.Visible = false; return end
    local angle = math.deg(math.atan2(diff.Y, diff.X))
    line.Visible = true
    line.AnchorPoint = Vector2.new(0.5, 0.5)
    line.Position = UDim2.fromOffset((from.X + to.X)/2, (from.Y + to.Y)/2)
    line.Size = UDim2.fromOffset(length, thickness)
    line.Rotation = angle
end

    return nil
end

local function UpdateESP(dt)
    if not ESP.Enabled then
        for _, d in pairs(ESPObjects) do HideAll(d) end
        return
    end
    local LocalChar = LocalPlayer.Character
    local alive = {}
    for _, Entity in CollectionService:GetTagged("Entity") do
        if Entity == LocalChar then continue end
        if not TeamCheck(Entity) then
            local d = ESPObjects[Entity]; if d then HideAll(d) end
            continue
        end
        local Player = Players:GetPlayerFromCharacter(Entity)
        local Humanoid = Entity:FindFirstChildOfClass("Humanoid")
        local HRP = Entity:FindFirstChild("HumanoidRootPart") or Entity:FindFirstChild("UpperTorso") or Entity:FindFirstChild("Torso")
        if not Humanoid or not HRP or Humanoid.Health <= 0 then
            local d = ESPObjects[Entity]; if d then HideAll(d) end
            continue
        end
        local Distance = (Camera.CFrame.Position - HRP.Position).Magnitude
        if ESP.LimitDistance and Distance > ESP.MaxDistance then
            local d = ESPObjects[Entity]; if d then HideAll(d) end
            continue
        end
        local RawTop, TopOn = WorldToScreen(HRP.Position + Vector3.new(0,3,0))
        local RawBottom, BotOn = WorldToScreen(HRP.Position - Vector3.new(0,2.5,0))
        if not TopOn and not BotOn then
            local d = ESPObjects[Entity]; if d then HideAll(d) end
            continue
        end
        local Data = ESPObjects[Entity] or CreateEntity(Entity)
        Data.Root.Visible = true
        alive[Entity] = true
        local Top, Bottom = RawTop, RawBottom
        local Height = math.max(math.abs(Bottom.Y - Top.Y), 4)
        local Width = math.max(Height * 0.55, 2)
        local CenterX = (Top.X + Bottom.X)/2
        local Left = CenterX - Width/2

        if Data.Highlight then
            Data.Highlight.Enabled = ESP.Highlight
            if ESP.Highlight then
                Data.Highlight.FillColor = ESP.HighlightFillColor
                Data.Highlight.FillTransparency = ESP.HighlightFillAlpha
                Data.Highlight.OutlineColor = ESP.HighlightOutlineColor
                Data.Highlight.OutlineTransparency = ESP.HighlightOutlineAlpha
            end
        end
        if Data.Box then
            if ESP.Box then
                Data.Box.Visible = true
                Data.Box.Position = UDim2.fromOffset(Left, Top.Y)
                Data.Box.Size = UDim2.fromOffset(Width, Height)
                Data.BoxStroke.Thickness = ESP.BoxThickness
                ApplyGradient(Data.BoxStrokeGradient, ESP, "Box", ESP.BoxSpin)
            else Data.Box.Visible = false end
        end
        if Data.Fill then
            if ESP.BoxFill then
                Data.Fill.Visible = true
                Data.Fill.Position = UDim2.fromOffset(Left, Top.Y)
                Data.Fill.Size = UDim2.fromOffset(Width, Height)
                Data.Fill.BackgroundTransparency = 1 - ESP.BoxFillAlpha
                ApplyGradient(Data.FillGradient, ESP, "BoxFill", ESP.BoxFillSpin)
            else Data.Fill.Visible = false end
        end
        if Data.Name then
            if ESP.Name then
                Data.Name.Visible = true
                Data.Name.Position = UDim2.fromOffset(CenterX - 90, Top.Y - 20)
                Data.Name.Size = UDim2.fromOffset(180, 16)
                ApplyGradient(Data.NameGradient, ESP, "Name", ESP.NameSpin)
                local name = Player and Player.Name or Entity.Name
                if ESP.NameShowHealth and Humanoid then
                    Data.Name.Text = string.format("%s <font color='rgb(180,180,180)'>[%d]</font>", name, math.floor(Humanoid.Health))
                else Data.Name.Text = name end
            else Data.Name.Visible = false end
        end
        if Data.Weapon then
            if ESP.Weapon and Player then
                local wpn = GetWeaponName(Player)
                if wpn then
                    Data.Weapon.Visible = true
                    Data.Weapon.Position = UDim2.fromOffset(CenterX - 90, Bottom.Y + 16)
                    Data.Weapon.Size = UDim2.fromOffset(180, 14)
                    ApplyGradient(Data.WeaponGradient, ESP, "Weapon", ESP.WeaponSpin)
                    Data.Weapon.Text = wpn
                else Data.Weapon.Visible = false end
            else Data.Weapon.Visible = false end
        end
        if Humanoid then
            local MaxH = Humanoid.MaxHealth > 0 and Humanoid.MaxHealth or 100
            local H = math.clamp(Humanoid.Health / MaxH, 0, 1)
            if Data.HealthBg and Data.HealthFill then
                if ESP.HealthBar then
                    Data.HealthBg.Visible = true
                    Data.HealthFill.Visible = true
                    Data.HealthBg.Position = UDim2.fromOffset(Left - ESP.HealthBarGap - ESP.HealthBarWidth, Top.Y)
                    Data.HealthBg.Size = UDim2.fromOffset(ESP.HealthBarWidth, Height)
                    Data.HealthFill.BackgroundColor3 = Color3.fromRGB(math.floor(255 * (1 - H)), math.floor(255 * H), 0)
                    Data.HealthFill.Position = UDim2.fromOffset(0, Height * (1 - H))
                    Data.HealthFill.Size = UDim2.fromOffset(ESP.HealthBarWidth, Height * H)
                else Data.HealthBg.Visible = false; Data.HealthFill.Visible = false end
            end
            if Data.HealthText then
                if ESP.HealthText then
                    Data.HealthText.Visible = true
                    Data.HealthText.Position = UDim2.fromOffset(Left - ESP.HealthBarGap - ESP.HealthBarWidth - 30, Top.Y + Height/2 - 8)
                    Data.HealthText.Size = UDim2.fromOffset(28, 16)
                    ApplyGradient(Data.HealthTextGradient, ESP, "HealthText", ESP.HealthTextSpin)
                    Data.HealthText.Text = tostring(math.floor(Humanoid.Health))
                else Data.HealthText.Visible = false end
            end
        end
        if Data.Distance then
            if ESP.Distance then
                Data.Distance.Visible = true
                Data.Distance.Position = UDim2.fromOffset(CenterX - 40, Bottom.Y + 2)
                Data.Distance.Size = UDim2.fromOffset(80, 14)
                ApplyGradient(Data.DistanceGradient, ESP, "Distance", ESP.DistanceSpin)
                Data.Distance.Text = string.format("[%d]", math.floor(Distance))
            else Data.Distance.Visible = false end
        end
        if Data.Skeleton then
            if ESP.Skeleton then
                for i, bone in ipairs(SkeletonBones) do
                    local Line = Data.Skeleton[i]
                    if Line then
                        local a = GetPartPosition(Entity, bone[1])
                        local b = GetPartPosition(Entity, bone[2])
                        if a and b then
                            local pa, oa = WorldToScreen(a)
                            local pb, ob = WorldToScreen(b)
                            if oa and ob then
                                PositionLine(Line, pa, pb, ESP.SkeletonThick)
                                ApplyGradient(Data.SkeletonGradients[i], ESP, "Skeleton", ESP.SkeletonSpin)
                            else Line.Visible = false end
                        else Line.Visible = false end
                    end
                end
                for i = #SkeletonBones + 1, #Data.Skeleton do Data.Skeleton[i].Visible = false end
            else
                for _, l in ipairs(Data.Skeleton) do l.Visible = false end
            end
        end
        if Data.Tracer then
            if ESP.Tracer then
                local V = Camera.ViewportSize
                local origin = Vector2.new(V.X * ESP.TracerOriginX, V.Y * ESP.TracerOriginY)
                PositionLine(Data.Tracer, origin, Vector2.new(CenterX, Bottom.Y), 1.5)
                ApplyGradient(Data.TracerGradient, ESP, "Tracer", ESP.TracerSpin)
            else Data.Tracer.Visible = false end
        end
    end
    for Ent, Data in pairs(ESPObjects) do
        if not alive[Ent] then HideAll(Data) end
    end
end

local aimbotHandler = { is_waiting = false, wait_end = 0, target = nil }
RunService.Heartbeat:Connect(function()
    if not Config.AimbotEnabled then
        aimbotHandler.is_waiting = false; aimbotHandler.target = nil; return
    end
    local useKeybind = not IsTouchDevice
    if useKeybind then
        if Config.AimbotKeybind and Config.AimbotKeybind ~= "" and Config.AimbotKeybind ~= "None" then
            local key = Enum.KeyCode[Config.AimbotKeybind]
            if key and not UserInputService:IsKeyDown(key) then return end
        end
    end
    local target = AimbotTarget()
    if not target then aimbotHandler.is_waiting = false; aimbotHandler.target = nil; return end
    if aimbotHandler.target ~= target then
        aimbotHandler.target = target
        aimbotHandler.is_waiting = true
        aimbotHandler.wait_end = os.clock() + (Config.AimbotReactionTime / 1000)
        return
    end
    if aimbotHandler.is_waiting then
        if os.clock() < aimbotHandler.wait_end then return end
        aimbotHandler.is_waiting = false
    end
    local TargetPos = target.Position
    local AimCFrame = CFrame.lookAt(Camera.CFrame.Position, TargetPos)
    local smoothing = math.clamp(Config.AimbotSmoothing / 100, 0.01, 1)
    local newCF = Camera.CFrame:Lerp(AimCFrame, smoothing)
    if CameraController and CameraController.MimicRotation then
        pcall(function() CameraController:MimicRotation(newCF) end)
    else Camera.CFrame = newCF end
end)

local triggerbotHandler = { is_shooting = false }
task.spawn(function()
    while task.wait() do
        if not Config.TriggerbotEnabled then triggerbotHandler.is_shooting = false; continue end
        local lf = FighterController and FighterController.LocalFighter
        if not (lf and lf:IsActive()) then continue end
        if Config.TriggerbotDisableOnScope and lf.EquippedItem and lf.EquippedItem:Get("IsAiming") then
            triggerbotHandler.is_shooting = false; continue
        end
        local hit = workspace:FindPartOnRayWithIgnoreList(
            Ray.new(Camera.CFrame.Position, Camera.CFrame.LookVector * 2500),
            {LocalPlayer.Character}
        )
        if hit and hit.Parent and hit.Parent:FindFirstChildWhichIsA("Humanoid")
            and hit.Parent:FindFirstChildWhichIsA("Humanoid").Health > 0 then
            if not TeamCheck(hit.Parent) then triggerbotHandler.is_shooting = false; continue end
            local target_player = Players:GetPlayerFromCharacter(hit.Parent)
            if not target_player then triggerbotHandler.is_shooting = false; continue end
            local whitelist = Config.TriggerbotWhitelistPart
            local passes = true
            if #whitelist > 0 and not table.find(whitelist, hit.Name) then passes = false end
            if Config.TriggerbotVisibleOnly and not HasLineOfSight(hit) then passes = false end
            if Config.TriggerbotIgnoreProtected and IsProtected(hit.Parent) then passes = false end
            if Config.TriggerbotAntiKatana and is_deflecting(target_player) then passes = false end
            if Config.TriggerbotRiotShield and has_riot(target_player) then passes = false end
            if passes then
                if not triggerbotHandler.is_shooting then
                    task.wait(Config.TriggerbotReactionTime)
                    triggerbotHandler.is_shooting = true
                end
                pcall(function() setthreadidentity(2) end)
                lf:Input("StartShooting")
                pcall(function() setthreadidentity(7) end)
                task.wait(Config.TriggerbotShootDelay)
            else triggerbotHandler.is_shooting = false end
        else triggerbotHandler.is_shooting = false end
    end
end)

local function RageUpdateDeflection()
    if not FighterController2 or not FighterController2.Objects then return end
    for _, fighterObj in FighterController2.Objects do
        local player = fighterObj.Player
        if not player then continue end
        if not fighterObj.Entity or not fighterObj.Entity:IsAlive() or fighterObj:Get("IsSpectating") then
            Ragebot._deflecting[player] = false
            continue
        end
        local equipped = fighterObj.EquippedItem
        local isKatana = equipped and equipped.ViewModel and equipped.ViewModel.Name == "Katana"
        local isDeflecting = false
        if isKatana then
            isDeflecting = (equipped._attack_cooldown and equipped._attack_cooldown > tick()) or false
        end
        Ragebot._deflecting[player] = isDeflecting
    end
end

local function RageIsEnemy(player)
    if player == lplr then return false end
    local duel = SpectateController.CurrentDuelSubject
    local localDueler = duel and duel:GetDueler(lplr)
    local localTeam = localDueler and localDueler:Get("TeamID") or nil
    if localTeam and duel and duel.Duelers then
        for _, dueler in duel.Duelers do
            if dueler.Player == player then
                local team = dueler:Get("TeamID")
                return team ~= localTeam
            end
        end
    end
    local pTeam = player:GetAttribute("TeamID")
    local lTeam = lplr:GetAttribute("TeamID")
    if pTeam and lTeam then
        return pTeam ~= lTeam
    end
    return true
end

local function RageHasForcefield(player)
    if not player or not player.Character then return false end
    return player.Character:FindFirstChildOfClass("ForceField") ~= nil
end

local function RageIsDeflecting(player)
    if not player then return false end
    return Ragebot._deflecting[player] == true
end

local function RageIsReloading()
    local lf = FighterController2 and FighterController2.LocalFighter
    if not lf or not lf.EquippedItem then return false end
    local result = false
    pcall(function()
        local item = lf.EquippedItem
        for _, key in ipairs({ "Reloading", "IsReloading", "IsReload" }) do
            local value = item:Get(key)
            if value == true or value == 1 or value == "true" then
                result = true
                break
            end
        end
    end)
    if result then return true end
    local startKey = Ragebot._reloadStart
    return startKey and (tick() - startKey) < 1.75
end

local function RageGetClosestTarget()
    local char = lplr.Character
    if not char then return nil, nil, nil end
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil, nil, nil end
    local closestPlayer, closestRoot, closestHead = nil, nil, nil
    local closestDist = 500
    for _, player in plrs:GetPlayers() do
        if not RageIsEnemy(player) then continue end
        local pChar = player.Character
        if not pChar then continue end
        local pRoot = pChar:FindFirstChild("HumanoidRootPart")
        local pHead = pChar:FindFirstChild("Head")
        local pHum = pChar:FindFirstChildWhichIsA("Humanoid")
        if not (pRoot and pHead and pHum and pHum.Health > 0) then continue end
        local dist = (myRoot.Position - pRoot.Position).Magnitude
        if dist < closestDist then
            closestDist = dist
            closestPlayer = player
            closestRoot = pRoot
            closestHead = pHead
        end
    end
    return closestPlayer, closestRoot, closestHead
end

local function RageHasKnifeViewModel(targetPlayer)
    if not targetPlayer then return false end
    local viewModels = ws:FindFirstChild("ViewModels")
    if not viewModels then return false end
    local targetName = targetPlayer.Name
    for _, model in viewModels:GetChildren() do
        if model:IsA("Model")
            and string.find(model.Name, targetName, 1, true)
            and string.find(model.Name, "Knife", 1, true) then
            return true
        end
    end
    return false
end

local function RageBetterRandom(mi, ma, dmi, dma)
    local val = math.random(mi, ma)
    repeat
        val = math.random(mi, ma)
    until val < dmi or val > dma
    return val
end

local function RageStartVoidHide()
    local char = lplr.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    Ragebot._hrp = hrp
    Ragebot._clientCF = hrp.CFrame
    Ragebot._clientVel = hrp.AssemblyLinearVelocity
    Ragebot._clientAngVel = hrp.AssemblyAngularVelocity
    Ragebot._hidden = true

    pcall(function()
        hrp.CFrame = CFrame.new(
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646),
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646),
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646)
        ) * CFrame.Angles(math.rad(math.pi), math.rad(math.pi), math.rad(math.pi))

        hrp.AssemblyLinearVelocity = Vector3.new(
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646),
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646),
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646)
        )

        hrp.AssemblyAngularVelocity = Vector3.new(
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646),
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646),
            RageBetterRandom(-2147483646, 2147483646, -1147483646, 1147483646)
        )
    end)

    if not Ragebot._csyncBound then
        Ragebot._csyncBound = true
        runS:BindToRenderStep("Ragebot_CSync", Enum.RenderPriority.First.Value, function()
            if Ragebot._hidden and Ragebot._hrp and Ragebot._clientCF then
                pcall(function()
                    Ragebot._hrp.CFrame = Ragebot._clientCF
                    Ragebot._hrp.AssemblyLinearVelocity = Ragebot._clientVel
                    Ragebot._hrp.AssemblyAngularVelocity = Ragebot._clientAngVel
                end)
            end
        end)
    end
end

local function RageStopVoidHide()
    if Ragebot._hrp and Ragebot._clientCF then
        pcall(function()
            Ragebot._hrp.CFrame = Ragebot._clientCF
        end)
    end
    Ragebot._hrp = nil
    Ragebot._clientCF = nil
    Ragebot._clientVel = nil
    Ragebot._clientAngVel = nil
    Ragebot._hidden = false
end

local function RageShouldHide(targetPlayer)
    if not targetPlayer then return true end
    if not FighterController2 or not FighterController2.LocalFighter then return true end
    if not lplr.Character or not lplr.Character:FindFirstChild("HumanoidRootPart") then return true end
    if RageHasForcefield(targetPlayer) then return true end
    if RageIsDeflecting(targetPlayer) then return true end
    if RageIsReloading() then return true end
    return false
end

local function RageIsMelee()
    local w = RageGetCurrentWeapon()
    if not w then return false end
    local melee = {
        ["Battle Axe"]=true, ["Chainsaw"]=true, ["Daggers"]=true, ["Fists"]=true,
        ["Gunblade"]=true, ["Katana"]=true, ["Knife"]=true,
        ["Scythe"]=true, ["Trowel"]=true,
    }
    return melee[w] == true or w:lower():find("melee") ~= nil
end

function RageGetCurrentWeapon()
    local viewModels = ws:FindFirstChild("ViewModels")
    if not viewModels then return nil end
    local firstPerson = viewModels:FindFirstChild("FirstPerson")
    if not firstPerson then return nil end
    for _, child in ipairs(firstPerson:GetChildren()) do
        local parts = {}
        for part in child.Name:gmatch("[^-]+") do
            table.insert(parts, part:match("^%s*(.-)%s*$"))
        end
        if #parts >= 2 then return parts[2] end
    end
    return nil
end

runS.Heartbeat:Connect(function()
    if not Ragebot.Enabled then
        if Ragebot._hidden then
            RageStopVoidHide()
        end
        return
    end

    RageUpdateDeflection()

    local targetPlayer, targetRoot, targetHead = RageGetClosestTarget()

    local shouldHide = RageShouldHide(targetPlayer)

    if shouldHide then
        if not Ragebot._hidden then
            RageStartVoidHide()
        end
        return
    end

    if Ragebot.VoidSpamEnabled then
        local now = tick()
        local elapsed = now - Ragebot._phaseStart
        if Ragebot._phase == "attack" then
            if elapsed >= Ragebot.VoidAttackTime then
                Ragebot._phase = "hide"
                Ragebot._phaseStart = now
                RageStartVoidHide()
                return
            end
        elseif Ragebot._phase == "hide" then
            if elapsed >= Ragebot.VoidHideTime then
                Ragebot._phase = "attack"
                Ragebot._phaseStart = now
                RageStopVoidHide()
            else
                return
            end
        end
    else
        if Ragebot._hidden then
            RageStopVoidHide()
        end
    end

    if not targetPlayer or not targetHead or not targetRoot then return end
    if not lplr.Character or not lplr.Character:FindFirstChild("HumanoidRootPart") then return end
    if not FighterController2 or not FighterController2.LocalFighter then return end
    local item = FighterController2.LocalFighter.EquippedItem
    if not item then return end
    if tick() - Ragebot._lastFire < Ragebot.FireRate then return end
    Ragebot._lastFire = tick()

    local melee = RageIsMelee()

    local desyncPos
    if RageHasKnifeViewModel(targetPlayer) then
        desyncPos = (targetRoot.CFrame * CFrame.new(0, 6, 0)).Position
    else
        desyncPos = (targetRoot.CFrame * CFrame.new(0, 1, 2)).Position
    end
    local desyncCF = CFrame.lookAt(desyncPos, targetHead.Position)

    local myRoot = lplr.Character:FindFirstChild("HumanoidRootPart")
    if myRoot then
        local oldCF = myRoot.CFrame
        local oldVel = myRoot.Velocity
        local oldRotVel = myRoot.RotVelocity
        myRoot.CFrame = desyncCF
        runS:BindToRenderStep("__rage_restore", 101, function()
            if myRoot then
                myRoot.CFrame = oldCF
                myRoot.Velocity = oldVel
                myRoot.RotVelocity = oldRotVel
            end
            runS:UnbindFromRenderStep("__rage_restore")
        end)
    end

    local originPos = desyncCF.Position
    local targetPos = targetHead.Position
    local aimCF = CFrame.lookAt(originPos, targetPos)
    local targetCF = targetHead.CFrame

    if melee then
        local randomOffset = Vector3.new(
            (math.random() - 0.5) * 0.1,
            (math.random() - 0.5) * 0.1,
            (math.random() - 0.5) * 0.1
        )
        local aimedPos = targetPos + randomOffset
        local objSpaceHeadOffset = targetHead.CFrame:ToObjectSpace(CFrame.new(aimedPos))
        local cameradata = {}
        cameradata[utf8.char(1)] = {
            [utf8.char(0)] = util:EncodeCFrame(aimCF),
            [utf8.char(1)] = util:EncodeCFrame(targetCF),
            [utf8.char(2)] = targetHead,
            [utf8.char(3)] = util:EncodeCFrame(objSpaceHeadOffset)
        }
        repS.Remotes.Replication.Fighter.UseItem:FireServer(
            item:Get("ObjectID"),
            enum:ToEnum("StartShooting"),
            cameradata,
            nil
        )
        return
    end

    local randomOffset = Vector3.new(
        (math.random() - 0.5) * 0.05,
        (math.random() - 0.5) * 0.05,
        (math.random() - 0.5) * 0.05
    )
    local aimedPos = targetHead.Position + randomOffset
    local objSpaceHeadOffset = targetHead.CFrame:ToObjectSpace(CFrame.new(aimedPos))
    local cameradata = {}
    cameradata[utf8.char(1)] = {
        [utf8.char(0)] = util:EncodeCFrame(aimCF),
        [utf8.char(1)] = util:EncodeCFrame(CFrame.lookAt(originPos, aimedPos)),
        [utf8.char(2)] = targetHead,
        [utf8.char(3)] = util:EncodeCFrame(objSpaceHeadOffset)
    }
    repS.Remotes.Replication.Fighter.UseItem:FireServer(
        item:Get("ObjectID"),
        enum:ToEnum("StartShooting"),
        cameradata,
        nil
    )
end)

local TracerTextureNames = {}

local TracerConfig = {
    Enabled = false, Texture = "Beam",
    ColorStart = WHITE, ColorMiddle = WHITE, ColorEnd = WHITE,
    Width0 = 0.15, Width1 = 0.15, Lifetime = 0.6, Glow = 1, Speed = 2, PositionLerp = 0.08,
}

local TracerActive = 0
local TracerMax = 64
local TracerList = {}

local function TracerCreate(from, to)
    if TracerActive >= TracerMax then return end
    TracerActive = TracerActive + 1
    local cfg = TracerConfig
    local direction = to - from
    local totalDist = direction.Magnitude
    if totalDist < 0.1 then TracerActive = TracerActive - 1; return end
    local unit = direction.Unit
    local originPart = Instance.new("Part")
    originPart.Anchored = true; originPart.CanCollide = false; originPart.CanTouch = false; originPart.CanQuery = false
    originPart.Transparency = 1; originPart.Size = Vector3.new(0.1,0.1,0.1); originPart.CFrame = CFrame.new(from); originPart.Parent = workspace
    local hitPart = Instance.new("Part")
    hitPart.Anchored = true; hitPart.CanCollide = false; hitPart.CanTouch = false; hitPart.CanQuery = false
    hitPart.Transparency = 1; hitPart.Size = Vector3.new(0.1,0.1,0.1); hitPart.CFrame = CFrame.new(from); hitPart.Parent = workspace
    local originAtt = Instance.new("Attachment"); originAtt.Parent = originPart
    local hitAtt = Instance.new("Attachment"); hitAtt.Parent = hitPart
    local beam = Instance.new("Beam")
    beam.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, cfg.ColorStart),
        ColorSequenceKeypoint.new(0.5, cfg.ColorMiddle),
        ColorSequenceKeypoint.new(1, cfg.ColorEnd),
    })
    beam.Brightness = cfg.Glow; beam.LightEmission = cfg.Glow; beam.LightInfluence = 0
    beam.TextureSpeed = cfg.Speed; beam.TextureLength = 3; beam.FaceCamera = true
    beam.Texture = TracerTextures[cfg.Texture] or ""; beam.TextureMode = Enum.TextureMode.Wrap
    beam.Attachment0 = originAtt; beam.Attachment1 = hitAtt; beam.Enabled = true
    beam.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.1), NumberSequenceKeypoint.new(0.5, 0), NumberSequenceKeypoint.new(1, 0.3) })
    beam.Width0 = cfg.Width0; beam.Width1 = cfg.Width1; beam.Parent = workspace
    local wallLine = Drawing.new("Line")
    wallLine.Thickness = math.max(cfg.Width0 * 12, 1.5); wallLine.Color = cfg.ColorStart
    wallLine.Transparency = 1; wallLine.Visible = false
    local elapsed = 0
    local conn = RunService.Heartbeat:Connect(function(dt)
        elapsed = elapsed + dt
        local lerpAlpha = math.clamp(elapsed / cfg.PositionLerp, 0, 1)
        local traveled = totalDist * lerpAlpha
        hitPart.CFrame = CFrame.new(from + unit * traveled)
        local a = Camera:WorldToViewportPoint(from)
        local b = Camera:WorldToViewportPoint(from + unit * traveled)
        if a.Z > 0 and b.Z > 0 then
            wallLine.From = Vector2.new(a.X, a.Y); wallLine.To = Vector2.new(b.X, b.Y)
            wallLine.Color = cfg.ColorStart; wallLine.Visible = true
        else wallLine.Visible = false end
        local alpha = TweenService:GetValue(elapsed / cfg.Lifetime, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
        beam.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.1 + alpha * 0.9), NumberSequenceKeypoint.new(0.5, alpha), NumberSequenceKeypoint.new(1, 0.3 + alpha * 0.7) })
        wallLine.Transparency = math.clamp(1 - alpha, 0, 1)
    end)
    table.insert(TracerList, { conn = conn, beam = beam, origin = originPart, hit = hitPart, wall = wallLine, start = tick() })
    task.delay(cfg.Lifetime, function()
        if conn then conn:Disconnect() end
        if wallLine then pcall(function() wallLine:Remove() end) end
        pcall(function() beam:Destroy() end); pcall(function() originPart:Destroy() end); pcall(function() hitPart:Destroy() end)
        TracerActive = math.max(TracerActive - 1, 0)
        for i, v in ipairs(TracerList) do if v.beam == beam then table.remove(TracerList, i); break end end
    end)
end

local function TracerCleanup()
    for _, t in ipairs(TracerList) do
        if t.conn then t.conn:Disconnect() end
        if t.wall then pcall(function() t.wall:Remove() end) end
        pcall(function() t.beam:Destroy() end); pcall(function() t.origin:Destroy() end); pcall(function() t.hit:Destroy() end)
    end
    TracerList = {}; TracerActive = 0
end

task.spawn(function()
    local ok, mod = pcall(function()
        return require(LocalPlayer.PlayerScripts:WaitForChild("Modules", 15):WaitForChild("TracerEffect", 15))
    end)
    if not ok or not mod or not mod.Play then return end
    local orig = mod.Play
    mod.Play = function(self, tracerData, config, extraData)
        if TracerConfig.Enabled and tracerData and tracerData.IsLocal and tracerData.RaycastResults then
            local muzzle = extraData and extraData.MuzzlePosition
            if muzzle then
                for _, hit in ipairs(tracerData.RaycastResults) do
                    if hit.Position then TracerCreate(muzzle, hit.Position) end
                end
            end
            return
        end
        return orig(self, tracerData, config, extraData)
    end
end)
